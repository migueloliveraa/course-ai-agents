```python
# Calculate the first 10,000 terms of the series 1 - 1/3 + 1/5 - 1/7 + ... and multiply by 4

num_terms = 10000
total = 0

for i in range(num_terms):
    term = 1 / (2 * i + 1)  # Generate the term 1/(2i + 1)
    if i % 2 == 0:  # If the index is even, add the term
        total += term
    else:           # If the index is odd, subtract the term
        total -= term

result = total * 4
print(result)
```
